if i want to run the same job multiple times with differnt inputs 
matrices 

### Without matrices
jobs: 
  test-java-17:
    steps:
    - name: Test Java 17    
      uses: actions/setup-java@v4
      with:
        java-version: '17'
    - name: Run Maven tests
      run: mvn test
  test-java-21:
    steps:
    - name: Test Java 21    
      uses: actions/setup-java@v4
      with:
        java-version: '21'
    - name: Run Maven tests
      run: mvn test
    
### with matrices
```yaml
jobs:
  tests:
     strategy:
       matrix:
         java-version: [ '17', '21' ]
     steps:
     - uses: actions/setup-java@v4
       with: 
         java-version: ${{ matrix.java-version }}
    - name: Run Maven tests
      run: mvn test
    # this will run 2 times    

```
